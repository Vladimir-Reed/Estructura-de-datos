using System;

/*
 * Tarea: Recorrido de matriz 3D (C#)
 */
class RecorridoMatriz3D
{
    static void Main(string[] args)
    {
        int[,,] matriz = {
            {
                {1, 2},
                {3, 4}
            },
            {
                {5, 6},
                {7, 8}
            }
        };

        int capas = matriz.GetLength(0);
        int filas = matriz.GetLength(1);
        int columnas = matriz.GetLength(2);

        for (int capa = 0; capa < capas; capa++)
        {
            Console.WriteLine("Capa " + capa + ":");
            for (int fila = 0; fila < filas; fila++)
            {
                for (int col = 0; col < columnas; col++)
                {
                    Console.Write(matriz[capa, fila, col] + "\t");
                }
                Console.WriteLine();
            }
        }
    }
}
